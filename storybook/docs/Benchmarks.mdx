export const Bars = ({ rows, value, unit, label }) => {
  const max = Math.max(...rows.map((r) => r[value]));
  return (
    <div style={{ margin: '12px 0 24px' }}>
      {rows.map((r) => {
        const rs = r.compiler.startsWith('stencilrs');
        return (
          <div key={r.compiler} style={{ margin: '8px 0' }}>
            <div style={{ fontSize: 13, marginBottom: 3 }}>{r.compiler}</div>
            <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
              <div style={{ height: 18, width: `${Math.max(0.6, (r[value] / max) * 100)}%`, background: rs ? '#e53e3e' : '#4299e1', borderRadius: 3 }} />
              <span style={{ fontVariantNumeric: 'tabular-nums', fontSize: 13, whiteSpace: 'nowrap' }}>
                {r[value].toLocaleString()} {unit}
              </span>
            </div>
          </div>
        );
      })}
      <div style={{ fontSize: 12, color: '#888' }}>{label}</div>
    </div>
  );
};

export const Table = ({ head, rows }) => (
  <table>
    <thead><tr>{head.map((h) => <th key={h}>{h}</th>)}</tr></thead>
    <tbody>{rows.map((r, i) => <tr key={i}>{r.map((c, j) => <td key={j}>{c}</td>)}</tr>)}</tbody>
  </table>
);
export const ms = (v) => `${v.toLocaleString()} ms`;
export const spread = (a) => `${a[0].toLocaleString()}–${a[1].toLocaleString()} ms`;

        <Table head={['Compiler', 'Wall (spawn to exit)', 'Spawn to build end', 'Wall, min–max']} rows={build.rows.map((r) => [r.compiler, ms(r.wallMedianMs), ms(r.buildEndMedianMs), spread(r.wallSpreadMs)])} />

## Build: a full `stencil build`, stencilrs swapped in

{data.stencilBuild.what}. Both rows run the same Stencil v5 CLI and config, alternating. In the

**With stencilrs: {data.stencilBuild.speedup} faster.** The gain is modest on purpose: only the component transform changes.
Stencil still runs its type-checked TypeScript pass over every file, processes the CSS, bundles the three
output targets and writes docs-json, and those dominate the build. This is a benchmark patch of a
throwaway copy of `@stencil/core` (`examples/ui-kit/bench-stencil-build.mjs`), not something stencilrs ships.

## What the numbers mean

- **Dev server and tests** compile files one at a time on demand: that's the per-file number. In the
  ui-kit's Vitest suite, Stencil's compiler took over 10 s to compile all 136 components for a single
  spec file's first test; stencilrs does it in a fraction of a second.
- **Production builds** spend time on more than compiling (bundling, CSS, minifying), so the speedup
  is smaller: ~3× through Vite. Inside a full `stencil build` it is ~1.1×, because Stencil's
  type-checked TypeScript pass, CSS and bundling remain Stencil's.
- stencilrs is single-threaded here; Vite calls it per file, so it parallelises with Vite's own work.

## Reproduce

```sh
cd parity && node bench.mjs 10                     # per file
cd examples/ui-kit && node bench-build.mjs 3       # whole ui-kit through Vite
cd examples/ui-kit && node bench-stencil-build.mjs 5 --json  # full stencil build, stock vs stencilrs
cd examples/ui-kit && node bench-cold.mjs 5        # cold start: fresh-process build, dev server to first response
```

To re-run all of it and rewrite this page's data and the headline numbers in the README, on a quiet
machine: `node scripts/refresh-benchmarks.mjs` (it writes `data/benchmarks.json`; the options are
listed at the top of the script).
